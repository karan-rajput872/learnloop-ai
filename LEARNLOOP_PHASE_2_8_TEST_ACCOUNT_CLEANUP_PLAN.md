## 4. Foreign Key Dependency & Safe Deletion Order

To prevent foreign-key violations during cleanup, child and dependent tables must be purged **before** parent records in `learners` and `users`:

```mermaid
flowchart TD
    subgraph Step 1: Purge Leaf & Child Tables
        C1[user_sessions & password_reset_tokens]
        C2[attempts & diagnostic_attempts]
        C3[chat_messages & revision_tasks]
        C4[concept_mastery & achievements]
        C5[user_activity_logs & user_activity_events]
        C6[user_streaks, gamification, settings, challenges]
        C7[login_events & owner_audit_logs]
    end

    subgraph Step 2: Purge Core Parent Entities
        P1[DELETE FROM users WHERE email NOT IN Curated]
        P2[DELETE FROM learners WHERE student_id NOT IN Curated]
    end

    Step 1 --> Step 2
```

### Executable Safe Cleanup Script (For Post-Approval Execution)
```sql
BEGIN;

-- Define curated identifiers:
-- Curated user IDs: ('usr_owner_main', 'usr_demo_teacher', 'usr_guest_demo')
-- Curated student IDs: ('student_owner', 'stu_demo_teacher', 'student_guest_demo', 'student_default')

-- 1. Purge child tables referencing student_id
DELETE FROM concept_mastery WHERE student_id NOT IN ('student_owner', 'stu_demo_teacher', 'student_guest_demo', 'student_default');
DELETE FROM attempts WHERE student_id NOT IN ('student_owner', 'stu_demo_teacher', 'student_guest_demo', 'student_default');
DELETE FROM diagnostic_attempts WHERE student_id NOT IN ('student_owner', 'stu_demo_teacher', 'student_guest_demo', 'student_default');
DELETE FROM revision_tasks WHERE student_id NOT IN ('student_owner', 'stu_demo_teacher', 'student_guest_demo', 'student_default');
DELETE FROM chat_messages WHERE student_id NOT IN ('student_owner', 'stu_demo_teacher', 'student_guest_demo', 'student_default');
DELETE FROM user_activity_logs WHERE student_id NOT IN ('student_owner', 'stu_demo_teacher', 'student_guest_demo', 'student_default');
DELETE FROM user_streaks WHERE student_id NOT IN ('student_owner', 'stu_demo_teacher', 'student_guest_demo', 'student_default');
DELETE FROM user_gamification WHERE student_id NOT IN ('student_owner', 'stu_demo_teacher', 'student_guest_demo', 'student_default');
DELETE FROM user_achievements WHERE student_id NOT IN ('student_owner', 'stu_demo_teacher', 'student_guest_demo', 'student_default');
DELETE FROM user_challenges WHERE student_id NOT IN ('student_owner', 'stu_demo_teacher', 'student_guest_demo', 'student_default');
DELETE FROM user_settings WHERE student_id NOT IN ('student_owner', 'stu_demo_teacher', 'student_guest_demo', 'student_default');
DELETE FROM user_notifications WHERE student_id NOT IN ('student_owner', 'stu_demo_teacher', 'student_guest_demo', 'student_default');

-- 2. Purge child tables referencing user_id
DELETE FROM user_sessions WHERE user_id NOT IN ('usr_owner_main', 'usr_demo_teacher', 'usr_guest_demo');
DELETE FROM password_reset_tokens WHERE user_id NOT IN ('usr_owner_main', 'usr_demo_teacher', 'usr_guest_demo');
DELETE FROM user_activity_events WHERE user_id NOT IN ('usr_owner_main', 'usr_demo_teacher', 'usr_guest_demo');
DELETE FROM login_events WHERE user_id NOT IN ('usr_owner_main', 'usr_demo_teacher', 'usr_guest_demo') AND email NOT IN ('karanrajput872@gmail.com', 'teacher@learnloop.ai', 'demo.learner@learnloop.ai');
DELETE FROM owner_audit_logs WHERE actor_id NOT IN ('usr_owner_main', 'usr_demo_teacher', 'usr_guest_demo') AND actor_email != 'karanrajput872@gmail.com';

-- 3. Purge parent tables
DELETE FROM users WHERE email NOT IN ('karanrajput872@gmail.com', 'teacher@learnloop.ai', 'demo.learner@learnloop.ai');
DELETE FROM learners WHERE student_id NOT IN ('student_owner', 'stu_demo_teacher', 'student_guest_demo', 'student_default');

-- 4. Re-align sequence for attempts table
SELECT setval('attempts_id_seq', COALESCE((SELECT MAX(id) FROM attempts), 1), true);

COMMIT;
```
